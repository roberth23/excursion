-- Impacto de la parametría en tipo_transaccion y medio de pago de ent_trx_loy
-- Compara el valor actual con el de sfc_tipos_trx_medio_pago (cruce por evento y concepto), por mes.
-- Una consulta por país. Volver a correr cuando se cierre la tarea de definición y documentación.

-- Colombia
WITH m AS (
  SELECT COALESCE(TRIM(CODEVENT),'') AS e, TRIM(CODCONCE) AS c, TIPO_TRANSACCION, MEDIO_PAGO
  FROM `fif-loy-co-datalake-prod.trf_loy_co_master_prd.sfc_tipos_trx_medio_pago`
)
SELECT
  DATE_TRUNC(t.trx_fch_mov, MONTH) AS mes,
  COALESCE(NULLIF(TRIM(t.tipo_transaccion),''),'(vacío)') AS tipo_actual,
  COALESCE(m.TIPO_TRANSACCION,'(sin fila en maestra)') AS tipo_esperado,
  COALESCE(NULLIF(TRIM(t.trx_mediopago),''),'(vacío)') AS medio_actual,
  COALESCE(m.MEDIO_PAGO,'(sin fila en maestra)') AS medio_esperado,
  COUNT(*) AS movimientos
FROM `fif-loy-co-datalake-prod.trf_loy_co_entities_prd.ent_trx_loy` t
LEFT JOIN m ON COALESCE(TRIM(t.trx_id_event),'') = m.e AND TRIM(t.trx_id_conc) = m.c
GROUP BY 1, 2, 3, 4, 5
ORDER BY 1, 2, 3, 4, 5;


-- Perú
WITH m AS (
  SELECT COALESCE(TRIM(CODEVENT),'') AS e, TRIM(CODCONCE) AS c, TIPO_TRANSACCION, MEDIO_PAGO
  FROM `pe-fif-fun-productivo.trf_loy_pe_master_prd.sfc_tipos_trx_medio_pago`
)
SELECT
  DATE_TRUNC(DATE(t.TRX_FCH_MOV), MONTH) AS mes,
  COALESCE(NULLIF(TRIM(t.TIPO_TRANSACCION),''),'(vacío)') AS tipo_actual,
  COALESCE(m.TIPO_TRANSACCION,'(sin fila en maestra)') AS tipo_esperado,
  COALESCE(NULLIF(TRIM(t.TRX_MEDIOPAGO),''),'(vacío)') AS medio_actual,
  COALESCE(m.MEDIO_PAGO,'(sin fila en maestra)') AS medio_esperado,
  COUNT(*) AS movimientos
FROM `pe-fif-fun-productivo.trf_loy_pe_prd.ent_trx_loy` t
LEFT JOIN m ON COALESCE(TRIM(t.TRX_ID_EVENT),'') = m.e AND TRIM(t.TRX_ID_CONC) = m.c
GROUP BY 1, 2, 3, 4, 5
ORDER BY 1, 2, 3, 4, 5;
