WITH co AS (
  SELECT 'CO' AS pais, COALESCE(TRIM(CODEVENT),'') AS e, TRIM(CODCONCE) AS c, COUNT(*) AS filas, COUNT(DISTINCT CONCAT(TIPO_TRANSACCION,'|',MEDIO_PAGO)) AS valores_distintos
  FROM `fif-loy-co-datalake-prod.trf_loy_co_master_prd.sfc_tipos_trx_medio_pago`
  GROUP BY 2, 3
),
pe AS (
  SELECT 'PE' AS pais, COALESCE(TRIM(CODEVENT),'') AS e, TRIM(CODCONCE) AS c, COUNT(*) AS filas, COUNT(DISTINCT CONCAT(TIPO_TRANSACCION,'|',MEDIO_PAGO)) AS valores_distintos
  FROM `pe-fif-fun-productivo.trf_loy_pe_master_prd.sfc_tipos_trx_medio_pago`
  GROUP BY 2, 3
)
SELECT 'duplicados_en_maestra' AS chk, pais, e AS evento, c AS concepto, filas AS n
FROM (SELECT * FROM co UNION ALL SELECT * FROM pe) WHERE filas > 1
UNION ALL
SELECT 'sin_fila_CO', 'CO', COALESCE(TRIM(t.trx_id_event),''), TRIM(t.trx_id_conc), COUNT(*)
FROM `fif-loy-co-datalake-prod.trf_loy_co_entities_prd.ent_trx_loy` t
LEFT JOIN co m ON COALESCE(TRIM(t.trx_id_event),'') = m.e AND TRIM(t.trx_id_conc) = m.c
WHERE m.c IS NULL GROUP BY 3, 4
UNION ALL
SELECT 'sin_fila_PE', 'PE', COALESCE(TRIM(t.TRX_ID_EVENT),''), TRIM(t.TRX_ID_CONC), COUNT(*)
FROM `pe-fif-fun-productivo.trf_loy_pe_prd.ent_trx_loy` t
LEFT JOIN pe m ON COALESCE(TRIM(t.TRX_ID_EVENT),'') = m.e AND TRIM(t.TRX_ID_CONC) = m.c
WHERE m.c IS NULL GROUP BY 3, 4
ORDER BY 1, 5 DESC;
